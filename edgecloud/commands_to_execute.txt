# 1. Go to project directory
cd ~/EdgeSense

# 2. Start Docker locally
sudo systemctl start docker

# 3. Start local edge service
docker compose -f edgecloud/docker-compose.yml up -d edge-ml-service

# 4. Verify local edge
docker ps

# 5. Test local edge
curl http://localhost:8001/

sudo docker ps

exit

# 9. Start decision engine locally
python3 edgecloud/decision_engine.py
